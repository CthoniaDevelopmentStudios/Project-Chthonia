// Protocol: Music-player
const deckPlayer = {
    queue: [], index: 0, audio: new Audio(), window: null,
    ERRORS: { 
        "READ": "FILE_IO_FAILURE", 
        "PLAY": "ENGINE_EXECUTION_ERR", 
        "QUEUE": "BUFFER_UNDERRUN", 
        "UI": "INTERFACE_RENDER_FAULT",
        "ENCODE": "BASE64_STREAM_CORRUPTION" 
    },
    
    // Broadcasts errors to the pack terminal
    reportErr: (key, line) => pack.report(`[CRITICAL:${deckPlayer.ERRORS[key]}] LINE:${line}`, 15),

    toB64: (buf) => {
        try {
            return btoa(new Uint8Array(buf).reduce((d, b) => d + String.fromCharCode(b), ''));
        } catch(e) { deckPlayer.reportErr("ENCODE", 16); }
    },
    
    // Visual Notif
    toast: (msg) => {
        const t = document.createElement("div");
        t.style = "position:fixed; top:20px; right:20px; background:#000; color:#0f0; border:1px solid #0f0; padding:12px; font-family:'Courier New', monospace; z-index:10000; box-shadow: 0 0 5px #0f0;";
        t.innerText = `[SYSTEM_NOTIF]: ${msg}`;
        document.body.appendChild(t);
        setTimeout(() => t.remove(), 3000);
    },

    // Strict Load Lock (prevents overlap)
    loadSong: (idx) => {
        deckPlayer.audio.pause();
        deckPlayer.audio.currentTime = 0;
        deckPlayer.audio.src = "data:audio/mp3;base64," + deckPlayer.queue[idx].data;
        deckPlayer.audio.play().catch(() => deckPlayer.reportErr("PLAY", 66));
    }
};

const buildDeckUI = () => {
    try {
        deckPlayer.window = document.createElement("div");
        deckPlayer.window.style = "position:fixed; bottom:10%; left:10%; width:600px; height:400px; background:#000; border:2px solid #0f0; color:#0f0; z-index:9000; display:grid; grid-template-columns: 1fr 1fr; font-family:'Courier New', monospace;";
        deckPlayer.window.innerHTML = `
            <div id="deck-list" style="border-right:1px solid #0f0; overflow-y:auto; padding:10px;"></div>
            <div style="padding:15px; display:flex; flex-direction:column; gap:10px;">
                <input type="file" id="f-in" style="background:#000; color:#0f0; border:1px solid #0f0;">
                <input id="s-n" placeholder="SONG_ID" style="background:#000; color:#0f0; border:1px solid #0f0;">
                <button id="add-b" style="background:#000; color:#0f0; border:1px solid #0f0; cursor:pointer;">CONVERT_&_ADD</button>
                <div style="margin-top:auto; text-align:center; display:flex; justify-content:center; gap:5px;">
                    <button id="p-b">⏮️</button><button id="play-b">▶️</button><button id="n-b">⏭️</button><button id="sh-b">🔀</button>
                </div>
                <button id="h-b">HIDE_WINDOW</button>
                <button id="w-b" style="color:#f00; border:1px solid #f00; background:#000; cursor:pointer;">UNDOCK_&_WIPE</button>
            </div>
        `;
        document.body.appendChild(deckPlayer.window);

        document.getElementById('add-b').onclick = () => {
            const file = document.getElementById('f-in').files[0];
            const reader = new FileReader();
            reader.onload = (e) => {
                deckPlayer.queue.push({ name: document.getElementById('s-n').value, data: deckPlayer.toB64(e.target.result) });
                document.getElementById('deck-list').innerHTML += `<div style="border-bottom:1px solid #333; padding:5px;">${deckPlayer.queue.length}. ${document.getElementById('s-n').value}</div>`;
            };
            reader.readAsArrayBuffer(file);
        };

        document.getElementById('play-b').onclick = () => {
            if (deckPlayer.queue.length === 0) return deckPlayer.reportErr("QUEUE", 63);
            if (deckPlayer.audio.paused) {
                deckPlayer.loadSong(deckPlayer.index);
                document.getElementById('play-b').innerText = "⏸️";
                deckPlayer.toast("PLAYING: " + deckPlayer.queue[deckPlayer.index].name);
            } else {
                deckPlayer.audio.pause();
                document.getElementById('play-b').innerText = "▶️";
            }
        };

        document.getElementById('n-b').onclick = () => {
            if (deckPlayer.queue.length === 0) return;
            deckPlayer.index = (deckPlayer.index + 1) % deckPlayer.queue.length;
            deckPlayer.loadSong(deckPlayer.index);
            deckPlayer.toast("NEXT: " + deckPlayer.queue[deckPlayer.index].name);
        };

        document.getElementById('p-b').onclick = () => {
            if (deckPlayer.queue.length === 0) return;
            deckPlayer.index = (deckPlayer.index - 1 + deckPlayer.queue.length) % deckPlayer.queue.length;
            deckPlayer.loadSong(deckPlayer.index);
        };

        document.getElementById('sh-b').onclick = () => {
            deckPlayer.queue.sort(() => Math.random() - 0.5);
            document.getElementById('deck-list').innerHTML = deckPlayer.queue.map((s,i) => `<div style="border-bottom:1px solid #333; padding:5px;">${i+1}. ${s.name}</div>`).join('');
            deckPlayer.toast("PLAYLIST SHUFFLED");
        };

        document.getElementById('h-b').onclick = () => deckPlayer.window.style.display = 'none';

        document.getElementById('w-b').onclick = () => {
            deckPlayer.audio.pause();
            deckPlayer.queue = [];
            deckPlayer.window.remove();
            deckPlayer.window = null;
            pack.report("DECK_CLEARED_SUCCESSFULLY", 0);
        };
        
        deckPlayer.audio.onended = () => document.getElementById('n-b').onclick();

    } catch(e) { deckPlayer.reportErr("UI", 94); }
};

const observer = new MutationObserver(() => {
    document.querySelectorAll('.pack-item').forEach(item => {
        if(item.textContent.includes(name) && !item.querySelector('.op-deck')) {
            const btn = document.createElement("button");
            btn.className = "op-deck";
            btn.innerText = "OPEN_DECK";
            btn.style = "background:#000; color:#0f0; border:1px solid #0f0; margin-left:5px; cursor:pointer;";
            btn.onclick = () => { if(!deckPlayer.window) buildDeckUI(); else deckPlayer.window.style.display = 'grid'; };
            item.appendChild(btn);
        }
    });
});
observer.observe(document.getElementById('pack-area'), { childList: true, subtree: true });
